fmt_n_pct <- function(n, total) {
  paste0(scales::comma(n), " (", scales::percent(n / total, accuracy = 0.1), ")")
}

tabla_frecuencia_gt <- function(data, var, n_top = 15, title = NULL) {
  v <- rlang::ensym(var)
  total <- nrow(data)

  tab <- data %>%
    mutate(.cat = as.character(!!v), .cat = if_else(is.na(.cat) | .cat == "", "NA", .cat)) %>%
    count(.cat, sort = TRUE, name = "n") %>%
    mutate(pct = n / sum(n)) %>%
    slice_head(n = n_top)

  gt::gt(tab) %>%
    gt::tab_header(title = title %||% glue::glue("Distribución de {rlang::as_name(v)}")) %>%
    gt::fmt_percent(columns = pct, decimals = 1) %>%
    gt::fmt_number(columns = n, decimals = 0, use_seps = TRUE) %>%
    gt::cols_label(.cat = rlang::as_name(v), n = "n", pct = "%")
}

cramers_v <- function(tbl) {
  if (any(dim(tbl) < 2)) return(NA_real_)
  chi <- suppressWarnings(chisq.test(tbl, correct = FALSE))
  n <- sum(tbl)
  k <- min(nrow(tbl) - 1, ncol(tbl) - 1)
  if (k <= 0 || n <= 0) return(NA_real_)
  sqrt(as.numeric(chi$statistic) / (n * k))
}

prueba_chi <- function(data, x, y) {
  x <- rlang::ensym(x)
  y <- rlang::ensym(y)

  d <- data %>%
    transmute(
      x = as.factor(!!x),
      y = as.factor(!!y)
    ) %>%
    filter(!is.na(x), !is.na(y))

  tbl <- table(d$x, d$y)
  if (any(dim(tbl) < 2)) {
    return(tibble(
      prueba = "No aplicable",
      estadistico = NA_real_,
      gl = NA_real_,
      p_value = NA_real_,
      cramer_v = NA_real_,
      n = nrow(d)
    ))
  }

  chi <- suppressWarnings(chisq.test(tbl, correct = FALSE))
  tibble(
    prueba = "Chi-cuadrado",
    estadistico = as.numeric(chi$statistic),
    gl = as.numeric(chi$parameter),
    p_value = as.numeric(chi$p.value),
    cramer_v = cramers_v(tbl),
    n = nrow(d)
  )
}

prueba_num_vs_cat <- function(data, y_num, x_cat) {
  y <- rlang::ensym(y_num)
  x <- rlang::ensym(x_cat)

  d <- data %>%
    transmute(
      y = as.numeric(!!y),
      x = as.factor(!!x)
    ) %>%
    filter(!is.na(y), !is.na(x))

  n_groups <- n_distinct(d$x)
  if (n_groups < 2) {
    return(tibble(
      prueba = "No aplicable",
      estadistico = NA_real_,
      p_value = NA_real_,
      n = nrow(d),
      grupos = n_groups
    ))
  }

  if (n_groups == 2) {
    w <- wilcox.test(y ~ x, data = d, exact = FALSE)
    return(tibble(
      prueba = "Wilcoxon rango-suma",
      estadistico = as.numeric(w$statistic),
      p_value = as.numeric(w$p.value),
      n = nrow(d),
      grupos = n_groups
    ))
  }

  k <- kruskal.test(y ~ x, data = d)
  tibble(
    prueba = "Kruskal-Wallis",
    estadistico = as.numeric(k$statistic),
    p_value = as.numeric(k$p.value),
    n = nrow(d),
    grupos = n_groups
  )
}

tabla_modelo_gt <- function(modelo, exponentiate = FALSE, title = "Resultados del modelo") {
  broom::tidy(modelo, conf.int = TRUE, exponentiate = exponentiate) %>%
    filter(term != "(Intercept)") %>%
    mutate(across(where(is.numeric), ~round(.x, 4))) %>%
    gt::gt() %>%
    gt::tab_header(title = title) %>%
    gt::cols_label(
      term = "Término",
      estimate = ifelse(exponentiate, "Estimación (OR)", "Estimación"),
      conf.low = "IC low",
      conf.high = "IC high",
      p.value = "p-valor"
    )
}

norm_geo <- function(x) {
  x <- as.character(x)
  x <- stringr::str_squish(x)
  y <- iconv(x, from = "", to = "ASCII//TRANSLIT")
  y[is.na(y)] <- x[is.na(y)]
  y <- toupper(stringr::str_squish(y))
  y
}

fmt_n_pct <- function(n, total) {
  paste0(scales::comma(n), " (", scales::percent(n / total, accuracy = 0.1), ")")
}

tabla_frecuencia_gt <- function(data, var, n_top = 15, title = NULL) {
  v <- rlang::ensym(var)
  total <- nrow(data)

  tab <- data %>%
    mutate(.cat = as.character(!!v), .cat = if_else(is.na(.cat) | .cat == "", "NA", .cat)) %>%
    count(.cat, sort = TRUE, name = "n") %>%
    mutate(pct = n / sum(n)) %>%
    slice_head(n = n_top)

  gt::gt(tab) %>%
    gt::tab_header(title = title %||% glue::glue("Distribución de {rlang::as_name(v)}")) %>%
    gt::fmt_percent(columns = pct, decimals = 1) %>%
    gt::fmt_number(columns = n, decimals = 0, use_seps = TRUE) %>%
    gt::cols_label(.cat = rlang::as_name(v), n = "n", pct = "%")
}

cramers_v <- function(tbl) {
  if (any(dim(tbl) < 2)) return(NA_real_)
  chi <- suppressWarnings(chisq.test(tbl, correct = FALSE))
  n <- sum(tbl)
  k <- min(nrow(tbl) - 1, ncol(tbl) - 1)
  if (k <= 0 || n <= 0) return(NA_real_)
  sqrt(as.numeric(chi$statistic) / (n * k))
}

prueba_chi <- function(data, x, y) {
  x <- rlang::ensym(x)
  y <- rlang::ensym(y)

  d <- data %>%
    transmute(
      x = as.factor(!!x),
      y = as.factor(!!y)
    ) %>%
    filter(!is.na(x), !is.na(y))

  tbl <- table(d$x, d$y)
  if (any(dim(tbl) < 2)) {
    return(tibble(
      prueba = "No aplicable",
      estadistico = NA_real_,
      gl = NA_real_,
      p_value = NA_real_,
      cramer_v = NA_real_,
      n = nrow(d)
    ))
  }

  chi <- suppressWarnings(chisq.test(tbl, correct = FALSE))
  tibble(
    prueba = "Chi-cuadrado",
    estadistico = as.numeric(chi$statistic),
    gl = as.numeric(chi$parameter),
    p_value = as.numeric(chi$p.value),
    cramer_v = cramers_v(tbl),
    n = nrow(d)
  )
}

prueba_num_vs_cat <- function(data, y_num, x_cat) {
  y <- rlang::ensym(y_num)
  x <- rlang::ensym(x_cat)

  d <- data %>%
    transmute(
      y = as.numeric(!!y),
      x = as.factor(!!x)
    ) %>%
    filter(!is.na(y), !is.na(x))

  n_groups <- n_distinct(d$x)
  if (n_groups < 2) {
    return(tibble(
      prueba = "No aplicable",
      estadistico = NA_real_,
      p_value = NA_real_,
      n = nrow(d),
      grupos = n_groups
    ))
  }

  if (n_groups == 2) {
    w <- wilcox.test(y ~ x, data = d, exact = FALSE)
    return(tibble(
      prueba = "Wilcoxon rango-suma",
      estadistico = as.numeric(w$statistic),
      p_value = as.numeric(w$p.value),
      n = nrow(d),
      grupos = n_groups
    ))
  }

  k <- kruskal.test(y ~ x, data = d)
  tibble(
    prueba = "Kruskal-Wallis",
    estadistico = as.numeric(k$statistic),
    p_value = as.numeric(k$p.value),
    n = nrow(d),
    grupos = n_groups
  )
}

tabla_modelo_gt <- function(modelo, exponentiate = FALSE, title = "Resultados del modelo") {
  broom::tidy(modelo, conf.int = TRUE, exponentiate = exponentiate) %>%
    filter(term != "(Intercept)") %>%
    mutate(across(where(is.numeric), ~round(.x, 4))) %>%
    gt::gt() %>%
    gt::tab_header(title = title) %>%
    gt::cols_label(
      term = "Término",
      estimate = ifelse(exponentiate, "Estimación (OR)", "Estimación"),
      conf.low = "IC low",
      conf.high = "IC high",
      p.value = "p-valor"
    )
}

agg_prov <- function(df, prov_col, dep_col = NULL) {
  d <- df
  if (!is.null(dep_col) && dep_col %in% names(d)) {
    d <- d %>% filter(norm_geo(.data[[dep_col]]) == "CUSCO")
  }
  d %>%
    mutate(prov_norm = norm_geo(.data[[prov_col]])) %>%
    filter(!is.na(prov_norm), prov_norm != "") %>%
    count(prov_norm, name = "n")

agg_dist <- function(df, prov_col, dist_col, dep_col = NULL) {

      dist_norm = norm_geo(.data[[dist_col]])

}

# Notas metodológicas finales

- Los p-valores deben interpretarse junto con tamaño de efecto (ej. **Cramér's V**) y contexto sustantivo.
- En muestras grandes, es habitual encontrar significancia estadística para efectos pequeños.
- En muestras pequeñas (educación superior/institutos), priorizar magnitud, dirección y plausibilidad sobre p-valor aislado.
- Para capítulo inferencial final, conviene agregar:
  - chequeos de supuestos por modelo,
  - análisis de sensibilidad de codificación de dependientes,
  - especificaciones alternativas.
